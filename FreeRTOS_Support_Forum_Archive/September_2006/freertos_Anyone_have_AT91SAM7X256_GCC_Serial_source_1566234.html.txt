<h3 class="thread-post-title">Anyone have AT91SAM7X256 GCC Serial source ?</h3>
<div class="thread-post-author">Posted by <a href="http://sourceforge.net/users/nobody">nobody</a> on September 4, 2006</div>
<div class="thread-post-content">#include &#8220;Board.h&#8221;<br />
#include &#8220;commd.h&#8221;
<p>//&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212; zamiast comm.c &#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8211; dodalem commd_&#8230;.<br />
//static xSemaphoreHandle debug_sem;</p>
<p>static char comm_initialized[3] = {0, 0, 0};</p>
<p>void commd_init(int port, int baud) {<br />
    switch (port) {<br />
        case USART0:<br />
            *AT91C_PIOA_PDR = AT91C_PA5_RXD0 |        /* Enable RxD0 Pin */<br />
                              AT91C_PA6_TXD0;         /* Enable TxD0 Pin */</p>
<p>            AT91C_BASE_US0-&gt;US_CR = AT91C_US_RSTRX |          /* Reset Receiver      */<br />
                                    AT91C_US_RSTTX |          /* Reset Transmitter   */<br />
                                    AT91C_US_RXDIS |          /* Receiver Disable    */<br />
                                    AT91C_US_TXDIS;           /* Transmitter Disable */</p>
<p>            AT91C_BASE_US0-&gt;US_MR = AT91C_US_USMODE_NORMAL |  /* Normal Mode */<br />
                  AT91C_US_CLKS_CLOCK    |  /* Clock = MCK */<br />
                  AT91C_US_CHRL_8_BITS   |  /* 8-bit Data  */<br />
                  AT91C_US_PAR_NONE      |  /* No Parity   */<br />
                  AT91C_US_NBSTOP_1_BIT;    /* 1 Stop Bit  */</p>
<p>            AT91C_BASE_US0-&gt;US_BRGR = MCK / 16 / baud;        /* Baud Rate Divisor */</p>
<p>            AT91C_BASE_US0-&gt;US_CR = AT91C_US_RXEN  |          /* Receiver Enable     */<br />
                                    AT91C_US_TXEN;            /* Transmitter Enable  */<br />
            comm_initialized[port] = 1;<br />
            break;</p>
<p>        case USART1:<br />
            *AT91C_PIOA_PDR = AT91C_PA21_RXD1 |        /* Enable RxD1 Pin */<br />
                              AT91C_PA22_TXD1;         /* Enable TxD1 Pin */</p>
<p>            AT91C_BASE_US1-&gt;US_CR = AT91C_US_RSTRX |          /* Reset Receiver      */<br />
                                    AT91C_US_RSTTX |          /* Reset Transmitter   */<br />
                                    AT91C_US_RXDIS |          /* Receiver Disable    */<br />
                                    AT91C_US_TXDIS;           /* Transmitter Disable */</p>
<p>            AT91C_BASE_US1-&gt;US_MR = AT91C_US_USMODE_NORMAL |  /* Normal Mode */<br />
                  AT91C_US_CLKS_CLOCK    |  /* Clock = MCK */<br />
                  AT91C_US_CHRL_8_BITS   |  /* 8-bit Data  */<br />
                  AT91C_US_PAR_NONE      |  /* No Parity   */<br />
                  AT91C_US_NBSTOP_1_BIT;    /* 1 Stop Bit  */</p>
<p>            AT91C_BASE_US1-&gt;US_BRGR = MCK / 16 / baud;        /* Baud Rate Divisor */</p>
<p>            AT91C_BASE_US1-&gt;US_CR = AT91C_US_RXEN  |          /* Receiver Enable     */<br />
                                    AT91C_US_TXEN;            /* Transmitter Enable  */<br />
            comm_initialized[port] = 1;<br />
            break;   </p>
<p>        case DUSART:<br />
            *AT91C_PIOA_PDR = AT91C_PA9_DRXD |         /* Enable RxD1 Pin */<br />
                              AT91C_PA10_DTXD;         /* Enable TxD1 Pin */</p>
<p>            AT91C_BASE_DBGU-&gt;DBGU_CR = AT91C_US_RSTRX |          /* Reset Receiver      */<br />
                                       AT91C_US_RSTTX |          /* Reset Transmitter   */<br />
                                       AT91C_US_RXDIS |          /* Receiver Disable    */<br />
                                       AT91C_US_TXDIS;           /* Transmitter Disable */</p>
<p>            AT91C_BASE_DBGU-&gt;DBGU_MR = AT91C_US_USMODE_NORMAL |  /* Normal Mode */<br />
                                       AT91C_US_CLKS_CLOCK    |  /* Clock = MCK */<br />
                                       AT91C_US_CHRL_8_BITS   |  /* 8-bit Data  */<br />
                                       AT91C_US_PAR_NONE      |  /* No Parity   */<br />
                                       AT91C_US_NBSTOP_1_BIT;    /* 1 Stop Bit  */</p>
<p>            AT91C_BASE_DBGU-&gt;DBGU_BRGR = MCK / 16 / baud;        /* Baud Rate Divisor */</p>
<p>            AT91C_BASE_DBGU-&gt;DBGU_CR = AT91C_US_RXEN  |          /* Receiver Enable     */<br />
                                       AT91C_US_TXEN;            /* Transmitter Enable  */<br />
            comm_initialized[port] = 1;<br />
            break;           <br />
       <br />
        default:<br />
            break;<br />
    }<br />
}</p>
<p>int commd_putc(int port,int ch) {<br />
    if (!comm_initialized[port]) return -3;<br />
    switch (port) {<br />
        case USART0:<br />
            while (!(AT91C_BASE_US0-&gt;US_CSR &amp; AT91C_US_TXRDY));   /* Wait for Empty Tx Buffer */<br />
            return (AT91C_BASE_US0-&gt;US_THR = ch);                 /* Transmit Character */<br />
           <br />
        case USART1:<br />
            while (!(AT91C_BASE_US1-&gt;US_CSR &amp; AT91C_US_TXRDY));   /* Wait for Empty Tx Buffer */<br />
            return (AT91C_BASE_US1-&gt;US_THR = ch);                 /* Transmit Character */<br />
           <br />
        case DUSART:<br />
            while (!(AT91C_BASE_DBGU-&gt;DBGU_CSR &amp; AT91C_US_TXRDY));   /* Wait for Empty Tx Buffer */<br />
            AT91C_BASE_DBGU-&gt;DBGU_THR = ch;                             /* Transmit Character */<br />
            return ch;               <br />
           <br />
        default:<br />
            break;<br />
    }<br />
    return -1;<br />
}</p>
<p>int commd_puts( int port, char* buff) {<br />
    if (!comm_initialized[port]) return -3;<br />
    while(*buff) {<br />
        if (*buff == &#8216;n&#8217;) commd_putc(port, &#8216;r&#8217;);<br />
            commd_putc(port, *buff++);<br />
    }<br />
    return 0;<br />
}</p>
<p>int commd_getc (int port) {<br />
    if (!comm_initialized[port]) return -3;<br />
    switch (port) {<br />
        case USART0:<br />
            while (!(AT91C_BASE_US0-&gt;US_CSR &amp; AT91C_US_RXRDY));   /* Wait for Full Rx Buffer */<br />
            return (AT91C_BASE_US0-&gt;US_RHR);                      /* Read Character */</p>
<p>        case USART1:<br />
            while (!(AT91C_BASE_US1-&gt;US_CSR &amp; AT91C_US_RXRDY));   /* Wait for Full Rx Buffer */<br />
            return (AT91C_BASE_US1-&gt;US_RHR);                      /* Read Character */<br />
           <br />
        case DUSART:<br />
            while (!(AT91C_BASE_DBGU-&gt;DBGU_CSR &amp; AT91C_US_RXRDY));   /* Wait for Full Rx Buffer */<br />
            return (AT91C_BASE_DBGU-&gt;DBGU_RHR);                      /* Read Character */<br />
           <br />
        default:<br />
            break;<br />
    }   <br />
    return -1;<br />
}</p>
<p>int commd_is_rx_empty(int port) {<br />
    if (!comm_initialized[port]) return -3;<br />
    switch (port) {<br />
        case USART0:<br />
            return (!(AT91C_BASE_US0-&gt;US_CSR &amp; AT91C_US_RXRDY));  /* Full Rx Buffer */<br />
        case USART1:<br />
            return (!(AT91C_BASE_US1-&gt;US_CSR &amp; AT91C_US_RXRDY));  /* Full Rx Buffer */<br />
        case DUSART:<br />
            return (!(AT91C_BASE_DBGU-&gt;DBGU_CSR &amp; AT91C_US_RXRDY));  /* Full Rx Buffer */   <br />
        default:<br />
            break;<br />
    }   <br />
    return -1;                   <br />
}</p>
<p>//&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;- PRINTF &#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;&#8212;-<br />
#include &lt;stdio.h&gt;                          /* I/O Functions */<br />
#include &lt;stdarg.h&gt;<br />
#include &#8220;FreeRTOS.h&#8221;<br />
#include &#8220;semphr.h&#8221;</p>
<p>static xSemaphoreHandle        printf_sem;</p>
<p>void kprintf_init() {<br />
    vSemaphoreCreateBinary( printf_sem );<br />
    while (printf_sem == NULL);    //halt if failed<br />
}</p>
<p>void kprintfFromISR(const char *format,&#8230;) {<br />
    static char buf[100];<br />
    va_list ap;<br />
    va_start(ap, format);<br />
    vsnprintf(buf, sizeof(buf), format, ap);    //viprintf() &#8211; without floating numbers. %f..<br />
    va_end(ap);   <br />
    commd_puts(DBG_DEV, buf);<br />
    //comm_puts(DBG_DEV, buf);<br />
}</p>
<p>void kprintf(const char *format,&#8230;) {<br />
    static char buf[100];<br />
    va_list ap;<br />
    va_start(ap, format);<br />
    vsnprintf(buf, sizeof(buf), format, ap);    //viprintf() &#8211; without floating numbers. %f..<br />
    va_end(ap);<br />
    while (!xSemaphoreTake(printf_sem, 100));                //block   <br />
    commd_puts(DBG_DEV, buf);<br />
    //comm_puts(DBG_DEV, buf);<br />
    xSemaphoreGive(printf_sem);                                //unblock<br />
}</p>
<p>void ppp_trace(int level, const char *format,&#8230;) {            // ppp debug<br />
    static char buf[100];<br />
    va_list ap;<br />
    va_start(ap, format);<br />
    vsnprintf(buf, sizeof(buf), format, ap);    //viprintf() &#8211; without floating numbers. %f..<br />
    va_end(ap);<br />
    while (!xSemaphoreTake(printf_sem, 100));                //block<br />
    commd_puts(DBG_DEV, buf);<br />
    //comm_puts(DBG_DEV, buf);<br />
    xSemaphoreGive(printf_sem);                                //unblock<br />
}</p>
