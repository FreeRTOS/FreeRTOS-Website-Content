~~~c
<h1>define NUM_UARTS 2</h1>
UART<em>HandleTypeDef uarts[NUM</em>UARTS];
&#8230;
void HAL<em>UART</em>MspInit(UART_HandleTypeDef *huart)
{
        if (huart == &amp;uarts[0]) {
                // Initialisation for first UART here
        } else if (huart == &amp;uarts[1]) {
                // Initialisation for second UART here
        } else if (&#8230;) {
        }
        // etc.
~~~
