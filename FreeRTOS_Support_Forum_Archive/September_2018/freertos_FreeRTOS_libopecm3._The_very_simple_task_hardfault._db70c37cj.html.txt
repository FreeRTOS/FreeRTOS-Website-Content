Startup( I don&#8217;t know actually where problem can be so I provide it too):
~~~
  .syntax unified
  .text
  .global memset<em>reg
  .type memset</em>reg, STT<em>FUNC
memset</em>reg:
  // call with the following (note that the arguments are not validated prior to use):
  // r0 &#8211; address of first word to write (inclusive)
  // r1 &#8211; address of first word following the address in r0 to NOT write (exclusive)
  // r2 &#8211; word value to be written
  // both addresses in r0 and r1 needs to be divisible by 4!
  .L<em>loop</em>begin:
    str r2, [r0], 4 // store the word in r2 to the address in r0, post-indexed
    cmp r0, r1
  bne .L<em>loop</em>begin
  bx lr
  .global reset<em>handler
  .type reset</em>handler, STT<em>FUNC
reset</em>handler:
  ldr r0, =<em>ram</em>start // r0 &#8211; point to beginning of SRAM
  ldr r1, =<em>ram</em>end   // r1 &#8211; point to byte after the end of SRAM
  ldr r2, =0          // r2 &#8211; the byte-sized value to be written
  bl memset_reg
  // copy .data section from flash to SRAM
  ldr r0, =<em>data          // dst addr
  ldr r1, =</em>data<em>loadaddr // src addr
  ldr r2, =</em>data_size     // length in bytes
  bl memcpy
  // enter the application code
  bl main
  // shutdown if the application code returns
  b shutdown
  .global shutdown
  .type shutdown, STT<em>FUNC
shutdown:
  cpsid f
  ldr r0, =0
  mov r1, r0
  mov r2, r0
  mov r3, r0
  mov r4, r0
  mov r5, r0
  mov r6, r0
  mov r7, r0
  mov r8, r0
  mov r9, r0
  mov r10, r0
  mov r11, r0
  mov r12, r0
  ldr lr, =0xffffffff
  ldr r0, =</em>ram<em>start
  ldr r1, =</em>ram<em>end
  // set to value in r2
  bl memset</em>reg
  b . // loop forever
  .ltorg // dump literal pool (for the ldr &#8230;,=&#8230; commands above)
~~~
