<pre><code>if( pxTCB == pxCurrentTCB )

        /* The task calling this function is querying its own state. */
        eReturn = eRunning;
    }

   I guess I can start by implementing the assembly version of eTaskState up to that point  &#8211; but I was wondering if there is any other quicker way to do this and if there&#8217;s some other function/macro someplace that does that.

</div></div><div class="thread-post"><h3 class="thread-post-title">Fast Task State Detection &#8211; Just &#8216;if running&#8217; detection</h3><div class="thread-post-author">Posted by <a href="http://sourceforge.net/users/westmorelandeng">westmorelandeng</a> on September 20, 2018</div><div class="thread-post-content">Hello All,
Well, I figured out a way to do this:
&#8230;
            mov xar7, dbl (<em>(#_save_xar7))                  ; save xar7
             mov xar6, dbl (</em>(#<em>save</em>xar6))                  ; save xar6
             amov #0x000000, xar7                            ; must be cleared
             amov #0x000000, xar6
                mov *sp(#1), ar7                                ;; sp+3
                   mov ar7, *(#_save_new_pxcode)
                    mov ssp, ar6
                    mov *ar6(#1), ar7                              ;; ssp+3
                     mov ar7, *(#_save_new_pxlcode)       ;  ==> now we have our new return address, and if a task, it&#8217;s where we are
<pre><code>                    mov dbl (*(#_pxCurrentTCB)), xar7
                    mov xar7, dbl(*(#_xCompareTCB))
                    mov dbl (*(#_save_xar7)), xar7                  ; restore xar7
                    mov dbl (*(#_save_xar6)), xar6                  ; restore xar6
</code></pre>
And:
            &#8230;
            .if 1
            mov dbl (*(_xCompareTCB)), xar6                              ; need to restore our return address<br />
                        mov xar7, ac0
                        mov xar6, ac1
                        CMPU AC1 != AC0, TC1 
                        BCC $5,TC1 
                        amov #0x000000, xar7
                        amov #0x000000, xar6
                        mov  *(#_save_new_pxcode), ar7
                        mov ar7, *sp(#0)
                        mov *(#_save_new_pxlcode) , ar7
                        mov ssp, ar6
                        mov ar7, *ar6(#0)
<pre><code>                    mov dbl (*(#_save_xar6)), xar6
                    .endif
</code></pre>
$5:
So, that&#8217;s not too bad.
