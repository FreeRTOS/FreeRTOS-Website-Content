<li>The semaphore/mutex macros determine whether or not to call the interrupt safe version.</li>

~~~

~~~
<h1>define GIVE<em>SEMAPHORE</em>FROM_ISR(s) do {         </h1>
  BaseType<em>t macroYieldRequest = pdFALSE;         
  BaseType</em>t macroStatus = xSemaphoreGiveFromISR((s), &amp;macroYieldRequest); 
  ASSERT<em>WARNING( macroStatus == pdTRUE );        
  portYIELD</em>FROM_ISR(macroYieldRequest);          
} while(0)
<h1>define GIVE<em>SEMAPHORE</em>THREAD(s) do {            </h1>
  BaseType<em>t macroStatus = xSemaphoreGive(s);    
  ASSERT</em>WARNING( macroStatus == pdTRUE );       
} while(0)
<h1>define GIVE_SEMAPHORE(s) do {  </h1>
  if( INTERRUPT<em>CONTEXT() )     
  {                             
     GIVE</em>SEMAPHORE<em>FROM</em>ISR(s);  
  }                             
  else                          
  {                             
     GIVE<em>SEMAPHORE</em>THREAD(s);    
  }                             
} while(0)
<h1>define TAKE<em>SEMAPHORE</em>FROM_ISR(s) do { </h1>
  BaseType<em>t macroYieldRequest = pdFALSE;         
  BaseType</em>t macroStatus = xSemaphoreTakeFromISR((s), &amp;macroYieldRequest); 
  ASSERT<em>WARNING( macroStatus == pdTRUE );        
  portYIELD</em>FROM_ISR(macroYieldRequest);          
} while(0)
<h1>define TAKE<em>SEMAPHORE</em>THREAD(s) do { </h1>
  BaseType<em>t macroStatus = xSemaphoreTake((s), portMAX</em>DELAY);    
  ASSERT_WARNING( macroStatus == pdTRUE ); 
} while(0)
<h1>define TAKE_SEMAPHORE(s) do {  </h1>
  if( INTERRUPT<em>CONTEXT() )     
  {                             
     ASSERT</em>REBOOT(FORCED);       
  }                             
  else                          
  {                             
     TAKE<em>SEMAPHORE</em>THREAD(s);    
  }                             
} while(0)
~~~

~~~
void Uart_Initialize(void)
{
  memset(&amp;uartObject, 0, sizeof(uartObject));    
  uartObject.tx.idle = 
    xSemaphoreCreateBinary();
  ASSERT_WARNING( uartObject.tx.idle != NULL );
  uartObject.tx.mutex = 
    xSemaphoreCreateMutex();
  ASSERT_WARNING( uartObject.tx.mutex != NULL );
  uartObject.pUartInstance = &#038;UART_INSTANCE;
  uartObject.errorTimerHandle =<br />
    xTimerCreate(&#8220;error timer&#8221;, 
                 MS<em>TO</em>TICKS(50),
                 ONE<em>SHOT</em>TIMER, 
                 &amp;uartObject,<br />
                 ErrorTimerCallback);
  InitializeUartDriver(&amp;uartObject, NRF<em>UART</em>BAUDRATE_115200);  
  PRINT(true, &#8220;Uart Initializedrn&#8221;);
}
~~~
