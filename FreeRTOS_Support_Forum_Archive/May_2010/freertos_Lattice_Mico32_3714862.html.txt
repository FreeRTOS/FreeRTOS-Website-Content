&#35;ifndef TIMER&#95;H<br />
&#35;define TIMER&#95;H<br />
&#35;include &lt;stdint.h&gt;  
&#35;define TIMER&#95;CONTROL&#95;INT&#95;BIT&#95;MASK&nbsp;&nbsp; 1 &lt;&lt; 0<br />
&#35;define TIMER&#95;CONTROL&#95;CONT&#95;BIT&#95;MASK&nbsp; 1 &lt;&lt; 1<br />
&#35;define TIMER&#95;CONTROL&#95;START&#95;BIT&#95;MASK 1 &lt;&lt; 2<br />
&#35;define TIMER&#95;CONTROL&#95;STOP&#95;BIT&#95;MASK&nbsp; 1 &lt;&lt; 3  
&#35;define TIMER&#95;BASE&#95;ADDRESS 0xA0000600 // This is the timer address for my configuration, it should be changed  
typedef struct &#123;<br />
&nbsp; volatile uint32&#95;t Status;<br />
&nbsp; volatile uint32&#95;t Control;<br />
&nbsp; volatile uint32&#95;t Period;<br />
&nbsp; volatile uint32&#95;t Snapshot;<br />
&#125; Timer&#95;t ;  
&#35;endif  
However, context switching does not occur. I can see that the timer interrupt handler is called, I can see that the systick is incrementing, but it doesn&#8217;t change context. Could someone explain what normally happens when the ISR handlers executes the following functions ?<br />
&nbsp; vTaskIncrementTick&#40;&#41;;<br />
&nbsp;&nbsp; vTaskSwitchContext&#40;&#41;;

</div></div><div class="thread-post"><h3 class="thread-post-title">Lattice Mico32</h3><div class="thread-post-author">Posted by <a href="http://sourceforge.net/users/jonenst">jonenst</a> on May 26, 2010</div><div class="thread-post-content">Ok, I had not seen the low level interrupt routine vPortInterruptHandler that saves and restores contexts… I was calling the high level IRQ directly. Now that I changed that, it works like a charm &#33;  
I had to add one line to port.c though, but maybe that&#8217;s because i&#8217;m running a simulated timer, not the real timer from lattice.   
Here&#8217;s the line :<br />
+&nbsp;&nbsp;&nbsp; pTimer-&gt;Period = configCPU&#95;CLOCK&#95;HZ / configTICK&#95;RATE&#95;HZ ;<br />
&nbsp;&nbsp;&nbsp;&nbsp; pTimer-&gt;Control = TIMER&#95;CONTROL&#95;START&#95;BIT&#95;MASK | TIMER&#95;CONTROL&#95;INT&#95;BIT&#95;MASK | TIMER&#95;CONTROL&#95;CONT&#95;BIT&#95;MASK;
